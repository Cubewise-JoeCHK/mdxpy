SELECT
{TM1SUBSETTOSET([}apqdimensionattributes].[}apqdimensionattributes],"Day in Month")} * {TM1SUBSETTOSET([}apqdimensionattributedefinitionmeasure].[}apqdimensionattributedefinitionmeasure],"Default")} DIMENSION PROPERTIES MEMBER_NAME ON 0,
{TM1SUBSETTOSET([}apqdimensionelements].[}apqdimensionelements],"Day in Month")} DIMENSION PROPERTIES MEMBER_NAME ON 1
FROM [}apqdimensionattributedefinition]
WHERE ([}apqdimensionhierarchies].[}apqdimensionhierarchies].[dayinmonth:dayinmonth])
