SELECT
{TM1SUBSETTOSET([msysversioninglog].[msysversioninglog],"Default")} DIMENSION PROPERTIES MEMBER_NAME ON 0,
{TM1SUBSETTOSET([sysversioningmodule].[sysversioningmodule],"Default")} DIMENSION PROPERTIES MEMBER_NAME ON 1
FROM [sysversioninglog]
WHERE ([scenario].[scenario].[act],[version].[version].[wip],[year].[year].[2018])
